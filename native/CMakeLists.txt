cmake_minimum_required(VERSION 3.22)
project(pocket_llama C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
option(POCKET_VULKAN_JNI "Build Vulkan server for an ART child process" OFF)
if(POCKET_VULKAN_JNI)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
add_link_options("-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../vendor/llama.cpp" upstream)
# Keep upstream server logic intact. Only replace its tiny executable entry point.
set_property(TARGET llama-server PROPERTY SOURCES "${CMAKE_CURRENT_LIST_DIR}/mobile_main.cpp")
if(POCKET_VULKAN_JNI)
    add_library(pocket_vulkan SHARED android_server_jni.cpp)
    target_link_libraries(pocket_vulkan PRIVATE llama-server-impl)
    target_compile_features(pocket_vulkan PRIVATE cxx_std_17)
    set_target_properties(pocket_vulkan PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    target_link_options(pocket_vulkan PRIVATE "-Wl,--exclude-libs,ALL")
endif()
